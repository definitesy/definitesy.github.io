#git常见命令

git init # 在当前文件夹初始化一个新的 Git 仓库
git clone <仓库地址> # 克隆远程仓库到本地
git status # 查看当前文件状态（改了哪些、哪些已暂存）
git add . # 添加当前文件夹所有改动到暂存区
git add <文件名> # 添加指定文件到暂存区
git commit -m “提交信息” # 提交暂存区内容到本地仓库
git commit -am “提交信息” # 跳过 add，直接提交所有已追踪文件的改动
git push # 推送当前分支到远程仓库
git push -u origin <分支名> # 第一次推送新分支，建立本地和远程的关联
git pull # 从远程拉取最新代码并合并到当前分支
git fetch # 从远程拉取最新代码，但不自动合并
git remote -v # 查看当前关联的远程仓库地址
git remote add origin <仓库地址> # 关联一个远程仓库
git remote set-url origin <新地址> # 修改远程仓库地址
git branch # 查看本地所有分支
git branch -a # 查看本地和远程所有分支
git branch <分支名> # 创建一个新分支
git checkout <分支名> # 切换到指定分支
git checkout -b <分支名> # 创建并切换到新分支
git switch <分支名> # 切换分支（Git 2.23+）
git switch -c <分支名> # 创建并切换到新分支（Git 2.23+）
git merge <分支名> # 把指定分支合并到当前分支
git branch -d <分支名> # 删除本地分支（已合并的）
git branch -D <分支名> # 强制删除本地分支（未合并也删）
git push origin –delete <分支名> # 删除远程分支
git log # 查看完整提交历史
git log –oneline # 查看精简提交历史（一行一个）
git log –oneline –graph # 图形化显示分支结构
git diff # 查看工作区未暂存的改动
git diff –staged # 查看暂存区和上次提交的区别
git show # 查看某次提交的详细信息
git restore <文件名> # 丢弃工作区的改动（还没 add 的）
git restore –staged <文件名> # 把文件从暂存区移出（取消 add）
git reset –soft HEAD1 # 撤销上一次 commit，改动保留在暂存区
git reset –mixed HEAD1 # 撤销上一次 commit，改动保留在工作区
git reset –hard HEAD~1 # 撤销上一次 commit，彻底丢弃改动（危险）
git revert # 用新提交来撤销某次提交（安全）
git stash # 临时保存所有未提交的改动
git stash pop # 恢复最近一次 stash 的改动
git stash list # 查看所有 stash 列表
git stash drop # 删除最近一次 stash
git cherry-pick # 把某次提交复制到当前分支
git rebase <分支名> # 把当前分支的提交变基到目标分支上
git reflog # 查看所有操作历史（找回丢失的提交）
git –version # 查看 Git 版本号
git help <命令名> # 查看某条命令的详细帮助文档
